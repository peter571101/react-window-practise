数据库 dorm_management
表     students
CREATE TABLE students (
    student_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '学生ID（自增主键）',
    name VARCHAR(20) NOT NULL COMMENT '姓名',
    password VARCHAR(20) NOT NULL COMMENT '密码',
    id_card CHAR(18) NOT NULL UNIQUE COMMENT '身份证号',
    gender ENUM('男', '女') NOT NULL COMMENT '性别',
    s_class VARCHAR(10) NOT NULL COMMENT '班级',
    building_number INT NOT NULL COMMENT '房间号',
    building_name VARCHAR(20) NOT NULL COMMENT '楼栋号',
    room_number INT NOT NULL COMMENT '床位号'
) COMMENT '学生住宿信息表';
